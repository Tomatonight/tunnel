cmake_minimum_required(VERSION 3.16)
project(tunnel LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(ANDROID AND TUNNEL_ANDROID_DEPS_ROOT)
    list(PREPEND CMAKE_PREFIX_PATH "${TUNNEL_ANDROID_DEPS_ROOT}/${ANDROID_ABI}")
    set(OPENSSL_ROOT_DIR "${TUNNEL_ANDROID_DEPS_ROOT}/${ANDROID_ABI}")
endif()

find_package(OpenSSL 3.0 REQUIRED COMPONENTS SSL Crypto)
find_package(Libevent CONFIG QUIET COMPONENTS core)
if(TARGET libevent::core)
    set(TUNNEL_EVENT_TARGET libevent::core)
elseif(TARGET libevent::core_shared)
    set(TUNNEL_EVENT_TARGET libevent::core_shared)
else()
    find_path(LIBEVENT_INCLUDE_DIR event2/event.h)
    find_library(LIBEVENT_LIBRARY NAMES event_core event)
    if(NOT LIBEVENT_INCLUDE_DIR OR NOT LIBEVENT_LIBRARY)
        message(FATAL_ERROR "libevent was not found; set CMAKE_PREFIX_PATH or LIBEVENT_INCLUDE_DIR/LIBEVENT_LIBRARY")
    endif()
    add_library(tunnel_event UNKNOWN IMPORTED)
    set_target_properties(tunnel_event PROPERTIES
        IMPORTED_LOCATION "${LIBEVENT_LIBRARY}"
        INTERFACE_INCLUDE_DIRECTORIES "${LIBEVENT_INCLUDE_DIR}")
    set(TUNNEL_EVENT_TARGET tunnel_event)
endif()

add_library(tunnel_core
    address.cpp crypto.cpp event.cpp gossip.cpp parse_config.cpp
    peer.cpp protocol.cpp tunnel.cpp udp_transport.cpp tls.cpp wire.cpp
    tcp_transport.cpp device.cpp kernel_device.cpp node.cpp address_service.cpp return_flows.cpp subnet_platform.cpp)
target_include_directories(tunnel_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(tunnel_core PUBLIC ${TUNNEL_EVENT_TARGET} OpenSSL::SSL OpenSSL::Crypto)
set_target_properties(tunnel_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(WIN32)
    target_compile_definitions(tunnel_core PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(tunnel_core PUBLIC ws2_32 iphlpapi)
endif()
if(MSVC)
    target_compile_options(tunnel_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(tunnel_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(ANDROID)
    add_library(tunnel_android SHARED android/jni.cpp)
    target_link_libraries(tunnel_android PRIVATE tunnel_core log)
else()
    add_executable(tunnel main.cpp)
    target_link_libraries(tunnel PRIVATE tunnel_core)
endif()
